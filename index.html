async function handleLogin(e){

async function handleJoin(e){

  e.preventDefault();

  const email = document.getElementById('joinEmail').value;

  const password = document.getElementById('joinPassword').value;

  const code = document.getElementById('joinCode').value.trim().toLowerCase();

  const btn = e.target.querySelector('button');

  btn.innerHTML = '<span class="loader"></span> Entrando...';

  btn.disabled = true;

  

  // Tenta logar primeiro

  let {data:authData, error:authError} = await sb.auth.signInWithPassword({email, password});

  if(authError){

    // Se não existe, cria

    const {data:newUser, error:signupError} = await sb.auth.signUp({

      email, password, options:{data:{name: email.split('@')[0]}}

    });

    if(signupError){btn.innerHTML='Entrar na família';btn.disabled=false;return showAuthMsg(signupError.message)}

    authData = newUser;

  }

  

  // Busca família pelo código

  const {data:family, error:famError} = await sb.from('families').select('id,name').eq('invite_code', code).single();

  if(famError || !family){btn.innerHTML='Entrar na família';btn.disabled=false;return showAuthMsg('Código de família inválido')}

  

  // Atualiza perfil do usuário

  const {error:updateError} = await sb.from('profiles').update({family_id: family.id, role:'colaborador'}).eq('id', authData.user.id);

  btn.innerHTML='Entrar na família';btn.disabled=false;

  if(updateError) return showAuthMsg('Erro ao entrar na família: '+updateError.message);

  

  await initApp();

}

// ===== INIT APP =====

async function initApp(){

  const {data:{session:s}} = await sb.auth.getSession();

  if(!s) return;

  session = s;

  

  // Busca perfil

  const {data:p, error} = await sb.from('profiles').select('*, families(name, invite_code)').eq('id', s.user.id).single();

  if(error || !p || !p.family_id){

    await sb.auth.signOut();

    showAuthMsg('Perfil não encontrado. Entre novamente.');

    return;

  }

  profile = p;

  familyId = p.family_id;

  

  document.getElementById('currentUser').textContent = p.name;

  document.getElementById('familyBadge').innerHTML = `<strong>${p.families.name}</strong><span>Código: ${p.families.invite_code.toUpperCase()}</span>`;

  

  document.getElementById('authScreen').style.display = 'none';

  document.getElementById('app').style.display = 'block';

  

  await loadAllData();

  renderAll();

}

    product: f.get('product'),

    value: parseFloat(f.get('value')),

    category_id: f.get('category_id')||null

  };

  let res;

  if(id){res = await sb.from('transactions').update(payload).eq('id', id)}

  else{res = await sb.from('transactions').insert(payload)}

async function importCSV(e,target){

  const file = e.target.files[0]; if(!file) return;

  Papa.parse(file,{

    header:true,skipEmptyLines:true,

    complete: async (res)=>{

      let rows = [];

      res.data.forEach(row=>{

        const data = row.Data || row.data || row.date;

        const prod = row.Produto || row.produto || row.product || row.Descricao;

        const val = row.Valor || row.valor || row.value;

        if(!data||!prod||!val) return;

        const date = data.includes('/')?parseDate(data.trim()):data.trim();

        const value = parseFloat(String(val).replace(',','.'));

        rows.push({date, product:prod, value});

      });

      if(target==='transaction'){

        const payload = rows.map(r=>({family_id:familyId,type:'expense',date:r.date,product:r.product,value:r.value,category_id:null}));

        const {error} = await sb.from('transactions').insert(payload);

        if(error) return alert('Erro: '+error.message);

      } else if(target==='card'){

        if(!data.cards.length) return alert('Cadastre um cartão primeiro');

        const cardId = prompt(`Para qual cartão?\n\n${data.cards.map(c=>`• ${c.name} (ID: ${c.id})`).join('\n')}\n\nCole o ID do cartão:`);

        const card = data.cards.find(c=>c.id===cardId);

        if(!card) return alert('Cartão não encontrado');

        const payload = rows.map(r=>{

          const d = new Date(r.date+'T00:00:00');

          const invoiceMonth = getInvoiceMonth(d, card.closing_day);

          return {card_id:card.id,family_id:familyId,date:r.date,product:r.product,value:r.value,invoice_month:invoiceMonth};

        });

        const {error} = await sb.from('card_charges').insert(payload);

        if(error) return alert('Erro: '+error.message);

      }

      alert(`${rows.length} registros importados!`);

      e.target.value='';

      await loadAllData();renderAll();

    }

  });

}

function getInvoiceMonth(date,closingDay){

  const d = new Date(date);

  if(d.getDate()>closingDay) d.setMonth(d.getMonth()+1);

  return d.toISOString().slice(0,7);

}

async function saveInvestment(e,id){

  e.preventDefault();

  const f = new FormData(e.target);

  const payload = {family_id:familyId,name:f.get('name'),type:f.get('type'),institution:f.get('institution'),amount:parseFloat(f.get('amount')),custom_rate:f.get('custom_rate')?parseFloat(f.get('custom_rate')):null,start_date:f.get('start_date'),end_date:f.get('end_date')||null};

  let res;

  if(id){res = await sb.from('investments').update(payload).eq('id', id)}

  else{res = await sb.from('investments').insert(payload)}

  if(res.error) return alert('Erro: '+res.error.message);

  closeModal();await loadAllData();renderAll();

}


async function deleteInvestment(id){

  if(!confirm('Excluir?'))return;

  await sb.from('investments').delete().eq('id', id);

  await loadAllData();renderAll();

}


function calcInvestment(inv){

  const type = INVESTMENT_TYPES.find(t=>t.id===inv.type);

  const rate = inv.custom_rate || type.rate;

  const start = new Date(inv.start_date);

  const end = inv.end_date ? new Date(inv.end_date) : new Date();

  const years = (end-start)/(1000*60*60*24*365.25);

  const future = inv.amount * Math.pow(1+rate/100, years);

  return {rate,years,future,profit:future-inv.amount};

}

function renderDashboard(){

  const currentMonth = new Date().toISOString().slice(0,7);

  const monthTx = data.transactions.filter(t=>t.date.startsWith(currentMonth));

  const income = monthTx.filter(t=>t.type==='income').reduce((s,t)=>s+t.value,0);

  const expense = monthTx.filter(t=>t.type==='expense').reduce((s,t)=>s+t.value,0);

  document.getElementById('totalIncome').textContent = fmt(income);

  document.getElementById('totalExpense').textContent = fmt(expense);

  document.getElementById('monthBalance').textContent = fmt(income-expense);

  document.getElementById('incomeCount').textContent = monthTx.filter(t=>t.type==='income').length+' transações';

  document.getElementById('expenseCount').textContent = monthTx.filter(t=>t.type==='expense').length+' transações';

  const wealth = data.accounts.reduce((s,a)=>s+a.balance,0)+data.investments.reduce((s,i)=>s+i.amount,0);

  document.getElementById('totalWealth').textContent = fmt(wealth);


  const years = [...new Set(data.transactions.map(t=>t.date.slice(0,4)))];

  const now = new Date().getFullYear().toString();

  if(!years.includes(now)) years.push(now);

  const yearSel = document.getElementById('chartYear');

  const curVal = yearSel.value;

  yearSel.innerHTML = years.sort().reverse().map(y=>`<option value="${y}" ${y===curVal?'selected':''}>${y}</option>`).join('');

  const year = yearSel.value || now;

  const months = ['Jan','Fev','Mar','Abr','Mai','Jun','Jul','Ago','Set','Out','Nov','Dez'];

  const incM = Array(12).fill(0), expM = Array(12).fill(0);

  data.transactions.filter(t=>t.date.startsWith(year)).forEach(t=>{

    const m = parseInt(t.date.slice(5,7))-1;

    if(t.type==='income') incM[m]+=t.value; else expM[m]+=t.value;

  });

  if(chartMonthly) chartMonthly.destroy();

  chartMonthly = new Chart(document.getElementById('chartMonthly'),{

    type:'bar',data:{labels:months,datasets:[{label:'Entradas',data:incM,backgroundColor:'#10b981'},{label:'Saídas',data:expM,backgroundColor:'#ef4444'}]},

    options:{responsive:true,maintainAspectRatio:false,plugins:{legend:{position:'bottom'}}}

  });


  const catData = {};

  monthTx.filter(t=>t.type==='expense').forEach(t=>{

    const c = data.categories.find(x=>x.id===t.category_id);

    const name = c?c.name:'Sem categoria';

    catData[name] = (catData[name]||0)+t.value;

  });

  const catColors = Object.keys(catData).map(n=>{const c=data.categories.find(x=>x.name===n);return c?c.color:'#94a3b8'});

  if(chartCategories) chartCategories.destroy();

  chartCategories = new Chart(document.getElementById('chartCategories'),{

    type:'doughnut',data:{labels:Object.keys(catData),datasets:[{data:Object.values(catData),backgroundColor:catColors}]},

    options:{responsive:true,maintainAspectRatio:false,plugins:{legend:{position:'bottom'}}}

  });


  const recent = [...data.transactions].slice(0,5);

  document.getElementById('recentTransactions').innerHTML = recent.length?recent.map(t=>`

    <div style="display:flex;justify-content:space-between;padding:10px 0;border-bottom:1px solid var(--border)">

      <div><div style="font-weight:500">${t.product}</div><div style="font-size:12px;color:var(--muted)">${fmtDate(t.date)}</div></div>

      <div style="font-weight:600;color:${t.type==='income'?'var(--success)':'var(--danger)'}">${t.type==='income'?'+':'-'} ${fmt(t.value)}</div>

    </div>

  `).join(''):'<div class="empty"><i class="fas fa-inbox"></i><div>Sem transações</div></div>';
